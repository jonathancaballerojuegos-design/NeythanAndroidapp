    <script>
        let ws;
        let audioCtx, analyser, dataArray, source;
        let isMicActive = false;
        let recognition;
        const coreImg = document.getElementById('core-img');
        const canvas = document.getElementById('waveCanvas');
        const ctx = canvas.getContext('2d');

        function setState(newState) {
            document.body.className = 'state-' + newState;
            document.getElementById('core-title').innerText = `N.E.Y.T.H.A.N. // ${newState}`;
        }

        function connectWS() {
            const protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
            ws = new WebSocket(`${protocol}//${window.location.host}/ws/neythan`);

            ws.onopen = () => {
                document.getElementById('telemetry-line').innerText = 'CPU: 12% | RED: SEGURA';
            };

            ws.onmessage = (event) => {
                const data = JSON.parse(event.data);
                if (data.type === 'telemetry') {
                    document.getElementById('telemetry-line').innerText =
                        `CPU: ${data.cpu}% | RAM: ${data.ram}% | RED: SEGURA`;
                } else if (data.type === 'status') {
                    setState(data.state);
                    document.getElementById('reply-box').innerText = data.message;
                } else if (data.type === 'response') {
                    setState(data.state);
                    document.getElementById('reply-box').innerText = data.reply;
                    speakText(data.reply);
                }
            };

            ws.onclose = () => {
                document.getElementById('telemetry-line').innerText = 'RECONECTANDO SERVIDOR...';
                setTimeout(connectWS, 3000);
            };
        }

        function speakText(text) {
            if ('speechSynthesis' in window) {
                window.speechSynthesis.cancel();
                const utterance = new SpeechSynthesisUtterance(text);
                utterance.lang = 'es-ES';
                utterance.rate = 1.02;
                utterance.pitch = 0.92;
                window.speechSynthesis.speak(utterance);
            }
        }

        async function initAudioVisualizer() {
            if (isMicActive) return;
            try {
                const stream = await navigator.mediaDevices.getUserMedia({
                    audio: { echoCancellation: true, noiseSuppression: true }
                });
                audioCtx = new (window.AudioContext || window.webkitAudioContext)();
                analyser = audioCtx.createAnalyser();
                analyser.fftSize = 128;
                source = audioCtx.createMediaStreamSource(stream);
                source.connect(analyser);
                dataArray = new Uint8Array(analyser.frequencyBinCount);
                isMicActive = true;
                drawWave();
            } catch (err) {
                document.getElementById('reply-box').innerText = 'Permiso de micrófono requerido, señor.';
            }
        }

        function drawWave() {
            requestAnimationFrame(drawWave);
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            if (!analyser) return;

            analyser.getByteFrequencyData(dataArray);
            let sum = 0;
            const barWidth = (canvas.width / dataArray.length) * 1.6;
            let x = 0;

            for (let i = 0; i < dataArray.length; i++) {
                const val = dataArray[i];
                sum += val;
                const barHeight = (val / 255) * (canvas.height * 0.85);
                ctx.fillStyle = `rgba(0, 240, 255, ${0.3 + val / 255})`;
                ctx.fillRect(x, (canvas.height - barHeight) / 2, barWidth - 2, barHeight);
                x += barWidth;
            }
            const avg = sum / dataArray.length;
            const pulse = 1 + (avg / 255) * 0.14;
            coreImg.style.setProperty('--pulse', pulse.toFixed(3));
        }

        function toggleListening() {
            initAudioVisualizer();
            const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
            if (!SpeechRecognition) {
                document.getElementById('reply-box').innerText = 'Reconocimiento de voz no soportado en este navegador.';
                return;
            }
            if (!recognition) {
                recognition = new SpeechRecognition();
                recognition.lang = 'es-ES';
                recognition.continuous = true;
                recognition.interimResults = false;

                recognition.onstart = () => {
                    setState('ESCUCHANDO');
                    document.getElementById('btn-mic').innerText = 'ESCUCHANDO...';
                };

                recognition.onresult = (event) => {
                    const last = event.results.length - 1;
                    const transcript = event.results[last][0].transcript.trim();
                    const lower = transcript.toLowerCase();

                    if (lower.includes('neythan') || lower.includes('despierta') || transcript.length > 2) {
                        const cleanCommand = transcript.replace(/neythan|despierta/gi, '').trim() || 'Reporta estado.';
                        document.getElementById('reply-box').innerText = `Usted: "${cleanCommand}"`;
                        if (ws && ws.readyState === WebSocket.OPEN) {
                            ws.send(JSON.stringify({ text: cleanCommand }));
                        }
                    }
                };

                recognition.onend = () => {
                    try { recognition.start(); } catch (e) {}
                };
            }
            try { recognition.start(); } catch (e) {}
        }

        connectWS();
    </script>
